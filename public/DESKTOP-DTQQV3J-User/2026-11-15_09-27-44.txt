  calcADX(_len) =>
      _up       = ta.change(high)
      _down     = -ta.change(low)
      _plusDM   = na(_up)   ? na : (_up > _down and _up > 0     ? _up   : 0)
      _minusDM  = na(_down) ? na : (_down > _up and _down > 0   ? _down : 0)
      _tr       = ta.rma(ta.tr(true), _len)
      _plus     = fixnan(100 * ta.rma(_plusDM,  _len) / _tr)
      _minus    = fixnan(100 * ta.rma(_minusDM, _len) / _tr)
      _sum      = _plus + _minus
      _adx      = 100 * ta.rma(math.abs(_plus - _minus) / (_sum == 0 ? 1 : _sum), _len)
      [_plus, _minus, _adx]
  
  calcADX_Masanakamura(_len) =>
      var float SmoothedTrueRange               = 0.0
      var float SmoothedDirectionalMovementPlus = 0.0
      var float SmoothedDirectionalMovementMinus= 0.0
      TrueRange                = math.max(math.max(high - low, math.abs(high - nz(close[1]))), math.abs(low - nz(close[1])))
      DirectionalMovementPlus  = high - nz(high[1]) > nz(low[1]) - low ? math.max(high - nz(high[1]), 0) : 0
      DirectionalMovementMinus = nz(low[1]) - low > high - nz(high[1]) ? math.max(nz(low[1]) - low, 0)   : 0
      SmoothedTrueRange                := nz(SmoothedTrueRange[1])                - (nz(SmoothedTrueRange[1])                / _len) + TrueRange
      SmoothedDirectionalMovementPlus  := nz(SmoothedDirectionalMovementPlus[1])  - (nz(SmoothedDirectionalMovementPlus[1])  / _len) + DirectionalMovementPlus
      SmoothedDirectionalMovementMinus := nz(SmoothedDirectionalMovementMinus[1]) - (nz(SmoothedDirectionalMovementMinus[1]) / _len) + DirectionalMovementMinus
      DIP  = SmoothedDirectionalMovementPlus  / SmoothedTrueRange * 100
      DIM  = SmoothedDirectionalMovementMinus / SmoothedTrueRange * 100
      DX   = math.abs(DIP - DIM) / (DIP + DIM) * 100
      adx  = ta.sma(DX, _len)
      [DIP, DIM, adx]

     void Init(const int l) { len = l; cnt = 0; sum = 0; val = 0; ready = false; }
     void Add(const double x)
     {
        if(!ready) { sum += x; cnt++; if(cnt >= len) { val = sum / len; ready = true; } }
        else        val = (val * (len - 1) + x) / len;
     }
  };
  
  struct CEma   // Pine ta.ema (seeded with first value)
  {
     double alpha, val;
     bool   ready;
     void Init(const int l) { alpha = 2.0 / (l + 1.0); val = 0; ready = false; }
     void Add(const double x)
     {
        if(!ready) { val = x; ready = true; }
        else        val = alpha * x + (1.0 - alpha) * val;
     }
  };

  double g_sTR = 0, g_sDMp = 0, g_sDMm = 0;
  double g_diP = 0, g_diM = 0, g_adx = 0;
  bool   g_adxOk = false;

  bool   g_sarBelow = false, g_sarOk = false;

  long   g_vwapDay = -1;
  double g_cumPV = 0, g_cumV = 0, g_vwapPrev = 0;
  bool   g_vwapPrevOk = false;
  double g_rsiV = 0, g_rsiVPrev = 0;
  bool   g_rsiVOk = false, g_rsiVPrevOk = false;

  //+------------------------------------------------------------------+
  //| Bar storage                                                      |
  //+------------------------------------------------------------------+
  void AppendBar(const MqlRates &r)
  {
     int n = g_n + 1;
     ArrayResize(g_o, n, 8192);  ArrayResize(g_h, n, 8192);
     ArrayResize(g_l, n, 8192);  ArrayResize(g_c, n, 8192);
     ArrayResize(g_v, n, 8192);  ArrayResize(g_dx, n, 8192);
     ArrayResize(g_t, n, 8192);
     g_o[g_n] = r.open;  g_h[g_n] = r.high;
     g_l[g_n] = r.low;   g_c[g_n] = r.close;
     g_v[g_n] = InpUseRealVolume ? (double)r.real_volume : (double)r.tick_volume;
     g_dx[g_n] = 0;
     g_t[g_n] = r.time;
     g_n = n;
  }
  
  bool Sma(const int len, double &out)
  {
     int i = g_n - 1;
     if(i < len - 1) return false;
     double s = 0;
     for(int k = i - len + 1; k <= i; k++) s += g_c[k];
     out = s / len;
     return true;
  }
  
  //+------------------------------------------------------------------+
  //| Indicators                                                       |
  //+------------------------------------------------------------------+
  void UpdatePivots(const int i)
  {
     if(i < InpLeft + InpRight) return;
     int p = i - InpRight;
     bool isH = true, isL = true;
     for(int k = p - InpLeft; k <= i; k++)
     {
        if(k == p) continue;
        if(g_h[k] > g_h[p]) isH = false;
        if(g_l[k] < g_l[p]) isL = false;
     }
     if(isH) { g_top = g_h[p]; g_hasTop = true; }
     if(isL) { g_bot = g_l[p]; g_hasBot = true; }
  }
  
  void UpdateAdx(const int i)
  {
     if(i < 1) return;
     double tr = MathMax(g_h[i] - g_l[i],
                 MathMax(MathAbs(g_h[i] - g_c[i - 1]), MathAbs(g_l[i] - g_c[i - 1])));
  
     if(InpAdxMode == ADX_CLASSIC)
     {
        double up = g_h[i] - g_h[i - 1];
        double dn = -(g_l[i] - g_l[i - 1]);
        double pdm = (up > dn && up > 0) ? up : 0;
        double mdm = (dn > up && dn > 0) ? dn : 0;
        g_rTr.Add(tr); g_rP.Add(pdm); g_rM.Add(mdm);
        if(g_rTr.ready && g_rTr.val > 0)
        {
           g_diP = 100.0 * g_rP.val / g_rTr.val;
           g_diM = 100.0 * g_rM.val / g_rTr.val;
           double sum = g_diP + g_diM;
           g_rAdx.Add(MathAbs(g_diP - g_diM) / (sum == 0 ? 1 : sum));
           if(g_rAdx.ready) { g_adx = 100.0 * g_rAdx.val; g_adxOk = true; }
        }
     }
     else // Masanakamura
     {
        double dmp = (g_h[i] - g_h[i - 1] > g_l[i - 1] - g_l[i]) ? MathMax(g_h[i] - g_h[i - 1], 0) : 0;
        double dmm = (g_l[i - 1] - g_l[i] > g_h[i] - g_h[i - 1]) ? MathMax(g_l[i - 1] - g_l[i], 0) : 0;
        g_sTR  = g_sTR  - g_sTR  / InpAdxLen + tr;
        g_sDMp = g_sDMp - g_sDMp / InpAdxLen + dmp;
        g_sDMm = g_sDMm - g_sDMm / InpAdxLen + dmm;
        if(g_sTR > 0)
        {
           g_diP = g_sDMp / g_sTR * 100.0;
           g_diM = g_sDMm / g_sTR * 100.0;
           double sum = g_diP + g_diM;
           g_dx[i] = (sum == 0) ? 0 : MathAbs(g_diP - g_diM) / sum * 100.0;
        }
        if(i >= InpAdxLen)
        {
           double s = 0;
           for(int k = i - InpAdxLen + 1; k <= i; k++) s += g_dx[k];
           g_adx = s / InpAdxLen;
           g_adxOk = true;
        }
     }
  }
  
  // Port of Pine ta.sar (reference implementation from the Pine docs)
  void UpdateSar(const int i)
  {
     if(i < 1) return;
     bool firstTrendBar = false;
     if(i == 1)
     {
        if(g_c[1] > g_c[0]) { g_sarBelow = true;  g_sarMaxMin = g_h[1]; g_sarRes = g_l[0]; }
        else                { g_sarBelow = false; g_sarMaxMin = g_l[1]; g_sarRes = g_h[0]; }
        firstTrendBar = true;
        g_sarAcc = InpSarStart;
     }
     g_sarRes = g_sarRes + g_sarAcc * (g_sarMaxMin - g_sarRes);
  
     if(g_sarBelow)
     {
        if(g_sarRes > g_l[i])
        {
           firstTrendBar = true; g_sarBelow = false;
           g_sarRes = MathMax(g_h[i], g_sarMaxMin);
           g_sarMaxMin = g_l[i]; g_sarAcc = InpSarStart;
        }
     }
     else
     {
        if(g_sarRes < g_h[i])
        {
           firstTrendBar = true; g_sarBelow = true;
           g_sarRes = MathMin(g_l[i], g_sarMaxMin);
           g_sarMaxMin = g_h[i]; g_sarAcc = InpSarStart;
        }
     }
  
     if(!firstTrendBar)
     {
        if(g_sarBelow)
        {
           if(g_h[i] > g_sarMaxMin) { g_sarMaxMin = g_h[i]; g_sarAcc = MathMin(g_sarAcc + InpSarInc, InpSarMax); }
        }
        else
        {
           if(g_l[i] < g_sarMaxMin) { g_sarMaxMin = g_l[i]; g_sarAcc = MathMin(g_sarAcc + InpSarInc, InpSarMax); }
        }
     }
  
     if(g_sarBelow)
     {
        g_sarRes = MathMin(g_sarRes, g_l[i - 1]);
        if(i > 1) g_sarRes = MathMin(g_sarRes, g_l[i - 2]);
     }
     else
     {
        g_sarRes = MathMax(g_sarRes, g_h[i - 1]);
        if(i > 1) g_sarRes = MathMax(g_sarRes, g_h[i - 2]);
     }
     g_sar = g_sarRes;
     g_sarOk = true;
  }
  
  // RSI( ta.vwap , len )  -  VWAP of hlc3 anchored to the day start
  void UpdateRsiVwap(const int i)
  {
     g_rsiVPrev = g_rsiV; g_rsiVPrevOk = g_rsiVOk;
  
     long day = ((long)g_t[i] - (long)InpServerUTCOffsetH * 3600) / 86400;
     if(day != g_vwapDay) { g_vwapDay = day; g_cumPV = 0; g_cumV = 0; }
     double hlc3 = (g_h[i] + g_l[i] + g_c[i]) / 3.0;
     g_cumPV += hlc3 * g_v[i];
     g_cumV  += g_v[i];
     double vwap = (g_cumV > 0) ? g_cumPV / g_cumV : hlc3;
  
     if(g_vwapPrevOk)
     {
        double chg = vwap - g_vwapPrev;
        g_rvU.Add(MathMax(chg, 0));
        g_rvD.Add(MathMax(-chg, 0));
        if(g_rvU.ready)
        {
           if(g_rvD.val == 0)      g_rsiV = 100.0;
           else if(g_rvU.val == 0) g_rsiV = 0.0;
           else                    g_rsiV = 100.0 - 100.0 / (1.0 + g_rvU.val / g_rvD.val);
           g_rsiVOk = true;
        }
     }
     g_vwapPrev = vwap; g_vwapPrevOk = true;
  }
  
  void UpdateRmi(const int i)
  {
     g_rmiPrev = g_rmi; g_rmiPrevOk = g_rmiOk;
     if(i < InpRmiMom) return;
     double up = MathMax(g_c[i] - g_c[i - InpRmiMom], 0);
     double dn = MathMax(g_c[i - InpRmiMom] - g_c[i], 0);
     g_eU.Add(up); g_eD.Add(dn);
     g_rmi = (g_eD.val == 0) ? 0.0 : 100.0 - 100.0 / (1.0 + g_eU.val / g_eD.val);
     g_rmiOk = true;
  }
  
  //+------------------------------------------------------------------+
  //| Process one CLOSED bar: indicators + Pine state machine          |
  //+------------------------------------------------------------------+
  void ProcessBar(const MqlRates &r)
  {
     AppendBar(r);
     const int i = g_n - 1;
     g_sigLong = false; g_sigShort = false;
  
     UpdatePivots(i);
     UpdateRsiVwap(i);
     UpdateRmi(i);
     UpdateSar(i);
     UpdateAdx(i);
     if(i < 1) return;
  
     //--- raw conditions ------------------------------------------------
     bool rsLong  = g_hasTop && g_c[i] > g_top;
     bool rsShort = g_hasBot && g_c[i] < g_bot;
  
     bool volOk = false;
     if(i >= InpVolLen - 1)
     {
        double s = 0;
        for(int k = i - InpVolLen + 1; k <= i; k++) s += g_v[k];
        volOk = g_v[i] > (s / InpVolLen) * InpVolMult;
     }
  
     bool Ladx = g_adxOk && g_diP > g_diM && g_adx > InpAdxTh;
     bool Sadx = g_adxOk && g_diP < g_diM && g_adx > InpAdxTh;
  
     double s1, s2, s3;
     bool maOk = Sma(InpSma1, s1) && Sma(InpSma2, s2) && Sma(InpSma3, s3);
     bool longMA  = maOk && s1 < g_c[i] && s2 < g_c[i] && s3 < g_c[i];
     bool shortMA = maOk && s1 > g_c[i] && s2 > g_c[i] && s3 > g_c[i];
  
     bool Lsar = g_sarOk && g_sar < g_c[i];
     bool Ssar = g_sarOk && g_sar > g_c[i];
  
     bool Lvap = g_rsiVOk && g_rsiVPrevOk && g_rsiVPrev <= InpRsiVwapOS && g_rsiV > InpRsiVwapOS;
     bool Svap = g_rsiVOk && g_rsiVPrevOk && g_rsiVPrev >= InpRsiVwapOB && g_rsiV < InpRsiVwapOB;
  
     bool Lrmi = g_rmiOk && g_rmiPrevOk && g_rmiPrev <= InpRmiOS && g_rmi > InpRmiOS;
     bool Srmi = g_rmiOk && g_rmiPrevOk && g_rmiPrev >= InpRmiOB && g_rmi < InpRmiOB;
  
     bool L1 = rsLong  && volOk && Ladx && longMA;
     bool S1 = rsShort && volOk && Sadx && shortMA;
     bool L2 = Lvap && Lsar;
     bool S2 = Svap && Ssar;
     bool L3 = Lrmi && Lsar && longMA;
     bool S3 = Srmi && Ssar && shortMA;
  
     //--- snapshot of previous-bar values (Pine [1]) --------------------
     const bool     pLongCond  = g_longCond,  pShortCond = g_shortCond;
     const int      pCiLong    = g_ciLong,    pCiShort   = g_ciShort;
     const datetime pLastLongTp = g_lastLongTp, pLastShortTp = g_lastShortTp;
     const datetime pLastLongSl = g_lastLongSl, pLastShortSl = g_lastShortSl;
     const int      pCiLongSl  = g_ciLongSl,  pCiShortSl = g_ciShortSl;
  
     g_longCond  = (L1 || L2 || L3);
     g_shortCond = (S1 || S2 || S3);
  
     g_ciLong  = pLongCond ? 1 : (pShortCond ? -1 : pCiLong);
     g_ciShort = pLongCond ? 1 : (pShortCond ? -1 : pCiShort);
  
     bool longCondition  = pLongCond  && pCiLong  == -1;
     bool shortCondition = pShortCond && pCiShort ==  1;
  
     if(longCondition)  { g_lastOpenLong  = g_c[i - 1]; g_lastLong  = g_t[i]; }
     if(shortCondition) { g_lastOpenShort = g_c[i - 1]; g_lastShort = g_t[i]; }
  
     bool inLong  = g_lastLong  > g_lastShort;
     bool inShort = g_lastShort > g_lastLong;
  
     const bool isLong  = (InpPosition != POS_SHORT_ONLY);
     const bool isShort = (InpPosition != POS_LONG_ONLY);
  
     //--- TP-1 -----------------------------------------------------------
     bool longTp  = isLong  && g_lastOpenLong  > 0 && g_h[i] > g_lastOpenLong  * (1.0 + InpTP1 / 100.0) && inLong;
     bool shortTp = isShort && g_lastOpenShort > 0 && g_l[i] < g_lastOpenShort * (1.0 - InpTP1 / 100.0) && inShort;
     if(longTp)  g_lastLongTp  = g_t[i];
     if(shortTp) g_lastShortTp = g_t[i];
  
     bool finalLongTp  = longTp  && g_lastLong  > pLastLongTp  && g_lastLong  > pLastLongSl;
     bool finalShortTp = shortTp && g_lastShort > pLastShortTp && g_lastShort > pLastShortSl;
  
     //--- SL -------------------------------------------------------------
     const double slf = InpSL / 100.0;
     bool longSl  = isLong  && g_lastOpenLong  > 0 && g_l[i] <= (1.0 - slf) * g_lastOpenLong
                    && !(g_o[i] < (1.0 - slf) * g_lastOpenLong);
     bool shortSl = isShort && g_lastOpenShort > 0 && g_h[i] >= (1.0 + slf) * g_lastOpenShort
                    && !(g_o[i] > (1.0 + slf) * g_lastOpenShort);
  
     bool sl0Long, sl0Short;
     if(InpPosition == POS_BOTH)
     {
        sl0Long  = longSl  && pCiLongSl  == -1 && !finalLongTp  && !shortCondition;
        sl0Short = shortSl && pCiShortSl == -1 && !finalShortTp && !longCondition;
     }
     else
     {
        sl0Long  = longSl  && pCiLongSl  == -1 && !finalLongTp;
        sl0Short = shortSl && pCiShortSl == -1 && !finalShortTp;
     }
  
     // last_xxx_sl uses the PREVIOUS bar's Final_xxx_sl (Pine var semantics)
     if(g_finalLongSl)  g_lastLongSl  = g_t[i];
     if(g_finalShortSl) g_lastShortSl = g_t[i];
  
     g_finalLongSl  = sl0Long  && g_lastLong  > pLastLongTp  && g_lastLong  > pLastLongSl;
     g_finalShortSl = sl0Short && g_lastShort > pLastShortTp && g_lastShort > pLastShortSl;
  
     g_ciLongSl  = (finalLongTp  || g_finalLongSl  || shortCondition) ? 1 : (longCondition  ? -1 : pCiLongSl);
     g_ciShortSl = (finalShortTp || g_finalShortSl || longCondition ) ? 1 : (shortCondition ? -1 : pCiShortSl);
  
     //--- re-entry after TP / SL ----------------------------------------
     if(finalLongTp  || g_finalLongSl)  g_ciLong  = -1;
     if(finalShortTp || g_finalShortSl) g_ciShort =  1;
  
     //--- signals --------------------------------------------------------
     g_sigLong = (g_longCond && !inLong)
              || (g_longCond && finalLongTp)
              || (g_longCond && g_finalLongSl)
              || (g_longCond && !longCondition && (g_lastLongTp > g_lastLong))
              || (g_longCond && !longCondition && (g_lastLongSl > g_lastLong));
  
     g_sigShort = (g_shortCond && !inShort)
               || (g_shortCond && finalShortTp)
               || (g_shortCond && g_finalShortSl)
               || (g_shortCond && !shortCondition && (g_lastShortTp > g_lastShort))
               || (g_shortCond && !shortCondition && (g_lastShortSl > g_lastShort));
  }

  int CountPos(const ENUM_POSITION_TYPE type)
  {
     int cnt = 0;
     for(int k = PositionsTotal() - 1; k >= 0; k--)
     {
        ulong tk = PositionGetTicket(k);
        if(tk == 0) continue;
        if(PositionGetString(POSITION_SYMBOL) != _Symbol) continue;
        if(PositionGetInteger(POSITION_MAGIC) != InpMagic) continue;
        if((ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE) == type) cnt++;
     }
     return cnt;
  }
  
  void CloseAll(const ENUM_POSITION_TYPE type)
  {
     for(int k = PositionsTotal() - 1; k >= 0; k--)
     {
        ulong tk = PositionGetTicket(k);
        if(tk == 0) continue;
        if(PositionGetString(POSITION_SYMBOL) != _Symbol) continue;
        if(PositionGetInteger(POSITION_MAGIC) != InpMagic) continue;
        if((ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE) != type) continue;
        g_trade.PositionClose(tk);
     }
  }
  
  double CalcVolume(const double price)
  {
     double step = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
     double vmin = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
     double vmax = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
     double vol;
     if(InpFixedLot > 0) vol = InpFixedLot;
     else
     {
        double cs = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_CONTRACT_SIZE);
        if(cs <= 0 || price <= 0) return 0;
        vol = AccountInfoDouble(ACCOUNT_EQUITY) * InpEquityPct / 100.0 / (price * cs);
     }
     vol = MathFloor(vol / step + 1e-9) * step;
     if(vol < vmin) { Print("Calculated volume below minimum lot: ", vol); return 0; }
     vol = MathMin(vol, vmax);
     int dig = (int)MathMax(0, MathCeil(-MathLog10(step)));
     return NormalizeDouble(vol, dig);
  }
  
  void OpenLegs(const bool isBuy)
  {
     // opposite direction is closed first (Pine strategy.entry reversal)
     CloseAll(isBuy ? POSITION_TYPE_SELL : POSITION_TYPE_BUY);
  
     const ENUM_POSITION_TYPE myType = isBuy ? POSITION_TYPE_BUY : POSITION_TYPE_SELL;
     double tps[3]; tps[0] = InpTP1; tps[1] = InpTP2; tps[2] = InpTP3;
  
     for(int k = 0; k < 3; k++)
     {
        if(CountPos(myType) >= InpMaxEntries) break;   // pyramiding limit
  
        double price = isBuy ? SymbolInfoDouble(_Symbol, SYMBOL_ASK)
                             : SymbolInfoDouble(_Symbol, SYMBOL_BID);
        double vol = CalcVolume(price);
        if(vol <= 0) break;
  
        double sl = isBuy ? price * (1.0 - InpSL / 100.0)    : price * (1.0 + InpSL / 100.0);
        double tp = isBuy ? price * (1.0 + tps[k] / 100.0)   : price * (1.0 - tps[k] / 100.0);
        sl = NormalizeDouble(sl, _Digits);
        tp = NormalizeDouble(tp, _Digits);
  
        string cmt = (isBuy ? "L_" : "S_") + IntegerToString(k + 1);
        bool ok = isBuy ? g_trade.Buy(vol, _Symbol, 0.0, sl, tp, cmt)
                        : g_trade.Sell(vol, _Symbol, 0.0, sl, tp, cmt);
        if(!ok)
           PrintFormat("Order %s failed: %d (%s)", cmt, g_trade.ResultRetcode(), g_trade.ResultRetcodeDescription());
     }
  }
  
  void ExecuteSignals(const datetime barTime)
  {
     if(barTime < InpStartTime || barTime > InpStopTime) return;
     const bool isLong  = (InpPosition != POS_SHORT_ONLY);
     const bool isShort = (InpPosition != POS_LONG_ONLY);
  
     if(g_sigShort && isShort)      OpenLegs(false);   // if both fire, short wins (same as Pine's final state)
     else if(g_sigLong && isLong)   OpenLegs(true);
  }
  
  //+------------------------------------------------------------------+
  //| Init / tick                                                      |
  //+------------------------------------------------------------------+
  int OnInit()
  {
     if(AccountInfoInteger(ACCOUNT_MARGIN_MODE) != ACCOUNT_MARGIN_MODE_RETAIL_HEDGING)
     {
        Print("This EA needs a HEDGING account (3 legs with different TPs).");
        return INIT_FAILED;
     }
     if(_Period != PERIOD_H4)
        Print("Warning: the strategy was designed for the 4h timeframe.");
  
     g_rTr.Init(InpAdxLen); g_rP.Init(InpAdxLen); g_rM.Init(InpAdxLen); g_rAdx.Init(InpAdxLen);
     g_rvU.Init(InpRsiVwapLen); g_rvD.Init(InpRsiVwapLen);
     g_eU.Init(InpRmiLen); g_eD.Init(InpRmiLen);
  
     g_trade.SetExpertMagicNumber(InpMagic);
     g_trade.SetDeviationInPoints(InpDeviation);
     g_trade.SetTypeFillingBySymbol(_Symbol);
  
     MqlRates rates[];
     ArraySetAsSeries(rates, false);
     int got = CopyRates(_Symbol, _Period, 1, InpWarmupBars, rates);
     if(got <= 0)
     {
        Print("CopyRates failed, history not ready: ", GetLastError());
        return INIT_FAILED;
     }
     for(int k = 0; k < got; k++) ProcessBar(rates[k]);
     g_lastProcessed = rates[got - 1].time;
     g_lastBarTime   = iTime(_Symbol, _Period, 0);
     return INIT_SUCCEEDED;
  }
  
  void OnTick()
  {
     datetime bt = iTime(_Symbol, _Period, 0);
     if(bt == 0 || bt == g_lastBarTime) return;
  
     MqlRates rates[];
     ArraySetAsSeries(rates, false);
     int got = CopyRates(_Symbol, _Period, g_lastProcessed + 1, bt - 1, rates);
     if(got <= 0) return;   // retry on next tick
